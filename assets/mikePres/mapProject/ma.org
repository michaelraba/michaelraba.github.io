<script>
  const superlayerIds = ["hDesk", "a1Desk", "agDesk"];
  let svgDoc = null;
  let superlayers = {};
  let currentSuperlayerId = null;

  document.addEventListener("DOMContentLoaded", () => {
    const obj = document.getElementById("mapObj");
    obj.addEventListener("load", () => {
      svgDoc = obj.contentDocument;

      // Get and store superlayer elements
      superlayerIds.forEach(id => {
        superlayers[id] = svgDoc.getElementById(id);
      });

      // Create superlayer buttons
      const superButtons = document.getElementById("superButtons");
      superlayerIds.forEach(id => {
        const btn = document.createElement("button");
        btn.textContent = id;
        btn.onclick = () => activateSuperlayer(id);
        superButtons.appendChild(btn);
      });

      // Activate first by default
      activateSuperlayer(superlayerIds[0]);
    });
  });

  function activateSuperlayer(id) {
    currentSuperlayerId = id;

    // Show selected superlayer, hide others
    for (const [sid, el] of Object.entries(superlayers)) {
      if (el) el.style.display = sid === id ? "inline" : "none";
    }

    // Create sublayer buttons
    const container = document.getElementById("subButtons");
    container.innerHTML = ""; // Clear old

    const group = superlayers[id];
    if (!group) return;

    const sublayerEls = Array.from(group.children).filter(el => el.id);
    sublayerEls.forEach(sub => {
      const btn = document.createElement("button");
      btn.textContent = sub.id;
      btn.onclick = () => showSublayer(sub.id);
      container.appendChild(btn);
    });
  }

  function showSublayer(subId) {
    const superlayer = superlayers[currentSuperlayerId];
    if (!superlayer) return;

    // Hide all sublayers
    Array.from(superlayer.children).forEach(el => {
      if (el.style) el.style.display = "none";
    });

    // Show selected sublayer
    const sub = svgDoc.getElementById(subId);
    if (sub && superlayer.contains(sub)) {
      sub.style.display = "inline";
      sub.querySelectorAll("*").forEach(el => {
        if (el.style) el.style.display = "inline";
      });
    }
  }
</script>
