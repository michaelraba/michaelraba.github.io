<script>
  let svgDoc = null;
  let superlayers = {};
  let currentSuperlayerId = 'hDesk';

  document.addEventListener("DOMContentLoaded", () => {
    const obj = document.getElementById("mapObj");

    obj.addEventListener("load", () => {
      svgDoc = obj.contentDocument;

      // Get references to both superlayers
      superlayers = {
        hDesk: svgDoc.getElementById("hDesk"),
        a1Desk: svgDoc.getElementById("a1Desk")
      };

      // Initially activate hDesk
      activateLayer("hDesk");
    });
  });

  function activateLayer(layerId) {
    currentSuperlayerId = layerId;

    for (const [id, group] of Object.entries(superlayers)) {
      if (!group) continue;
      group.style.display = (id === layerId) ? "inline" : "none";
    }
  }

  function showSub(subId) {
    const activeLayer = superlayers[currentSuperlayerId];
    if (!activeLayer) return;

    // Hide all direct children of the current superlayer
    Array.from(activeLayer.children).forEach(child => {
      if (child.style) child.style.display = "none";
    });

    // Try to show the specified sublayer
    const sub = svgDoc.getElementById(subId);
    if (sub && activeLayer.contains(sub)) {
      sub.style.display = "inline";
      sub.querySelectorAll("*").forEach(el => {
        if (el.style) el.style.display = "inline";
      });
    } else {
      console.warn(`Sublayer "${subId}" not found in "${currentSuperlayerId}"`);
    }
  }
</script>
